# Preferencias de uso del contenido para sistemas de IA (Content Signals,
# https://contentsignals.org/). No son reglas de rastreo: declaran qué usos
# consentimos sobre el contenido ya rastreado.
#   search=yes    -> indexar y enlazar en buscadores, incluidos los de IA.
#   ai-input=yes  -> usarlo como fuente citada al responder (RAG/grounding).
#                    Es exactamente para lo que existe este directorio.
#   ai-train=no   -> no usarlo como material de entrenamiento de modelos.
Content-Signal: search=yes, ai-input=yes, ai-train=no

# /r es el interstitial de salida: es noindex y multiplica URLs por
# querystring (?t=&p=&l=). Enlazarlo es necesario, rastrearlo no aporta nada.
Disallow: /r
Disallow: /r?
